class Solution {
    int countStrings(int n) {
       return count(0,0,n);
    }
    int count(int idx,int prev,int n){
        if(idx==n){
            return 1;
        }
        if(prev==1){
            return count(idx+1,0,n);
        }
        return count(idx+1,0,n)+count(idx+1,1,n);
    }
}
